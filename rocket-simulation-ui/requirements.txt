# JARVIS Rocket Simulation - dependencies
#
# Direct dependencies only. This used to be a full `pip freeze` dump with every
# transitive package pinned to an exact version, which meant the install broke
# on any Python newer than the one it was frozen on (pandas 2.3.2 and
# PyQt5_sip 12.17.0 in particular have no wheels for CPython 3.14, so pip fell
# back to compiling them from source and failed). pillow, fonttools, kiwisolver,
# contourpy, python-dateutil and friends are pulled in automatically - they do
# not belong here.
#
# Lower bounds are the earliest releases that ship standard CPython 3.14
# Windows wheels, so this installs without a compiler on Python 3.12, 3.13
# and 3.14. Upper bounds guard against the next major version changing APIs.

PyQt5>=5.15.11,<6
matplotlib>=3.10.5,<4
numpy>=2.3.2,<3
scipy>=1.16.1,<2

# Only used by the console helper simulation.plot_table_and_stats(); the GUI
# itself does not import pandas.
pandas>=2.3.3,<4
